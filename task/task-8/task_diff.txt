+import { useState } from 'react'
+import { validateTaskForm } from '../utils/taskValidation'
+
+/**
+ * Custom hook for managing task form data
+ * @param {string} listId - Default list ID
+ * @returns {Object} Task data and handlers
+ */
+export const useTaskData = (listId) => {
+  const [formData, setFormData] = useState({
+    title: '',
+    description: '',
+    status: 'todo',
+    priority: 'medium',
+    dueDate: '',
+    assignee: '',
+    list: listId || '',
+    tags: []
+  })
+
+  const [errors, setErrors] = useState({})
+
+  const handleChange = (e) => {
+    const { name, value } = e.target
+    setFormData(prev => ({ ...prev, [name]: value }))
+    // Clear error for this field
+    if (errors[name]) {
+      setErrors(prev => ({ ...prev, [name]: null }))
+    }
+  }
+
+  const validate = () => {
+    const validationErrors = validateTaskForm(formData)
+    setErrors(validationErrors)
+    return Object.keys(validationErrors).length === 0
+  }
+
+  return {
+    formData, 
+    errors,
+    handleChange,
+    validate
+  }
+}

+/**
+ * Validates task form data
+ * @param {Object} formData - Task form data to validate
+ * @returns {Object} Object containing validation errors
+ */
+export const validateTaskForm = (formData) => {
+  const errors = {}
+
+  if (!formData.title || !formData.title.trim()) {
+    errors.title = 'Task title is required'
+  }
+
+  if (formData.title && formData.title.length > 200) {
+    errors.title = 'Task title must be less than 200 characters'
+  }
+
+  if (!formData.list) {
+    errors.list = 'Please select a list'
+  }
+
+  if (formData.dueDate) {
+    const dueDate = new Date(formData.dueDate)
+    if (dueDate < new Date()) {
+      errors.dueDate = 'Due date cannot be in the past'
+    }
+  }
+
+  return errors
+}
